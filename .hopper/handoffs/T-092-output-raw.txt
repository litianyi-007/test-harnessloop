   147	def dig(obj: Any, *path: str) -> Any:
   148	    """Safely walk a chain of dict keys; returns None as soon as something isn't a dict."""
   149	    cur = obj
   150	    for p in path:
   151	        if isinstance(cur, dict):
   152	            cur = cur.get(p)
   153	        else:
   154	            return None
   155	    return cur

   158	def is_valid_seq(v: Any) -> bool:
   159	    """seq must be a genuine int -- no coercion.
   160	
   161	    2026-08-11 second-round fix: the previous normalize_seq() let a JSON
   162	    boolean through unchanged, which is a real hole because in Python
   163	    `True == 1` and `hash(True) == hash(1)` -- so a (id, True) key from one
   164	    source and a (id, 1) key from the other would silently collide in a
   165	    dict/set and be treated as "matched", even though a JSON `true` on the
   166	    wire and a JSON `1` in history are not the same value at all. A plain
   167	    `isinstance(v, int)` check alone does NOT catch this, because bool is a
   168	    subclass of int in Python -- the bool exclusion has to be explicit.
   169	    Floats are rejected too (the old code's `int(v)` would silently
   170	    floor-truncate e.g. 2.7 -> 2, which could just as easily mask a genuine
   171	    type mismatch instead of a harmless representation difference), and so is
   172	    any other type (str included -- no more "best-effort" parsing of numeric
   173	    strings). This mirrors is_valid_id() below: both halves of a key must now
   174	    pass a real type check, not just "is not None".
   175	    """
   176	    return isinstance(v, int) and not isinstance(v, bool)
   177	
   178	
   179	def is_valid_id(v: Any) -> bool:
   180	    """id must be a non-empty str -- no coercion from int or any other type
   181	    (2026-08-11 second-round fix; see is_valid_seq() above for the matching
   182	    seq-side rationale)."""
   183	    return isinstance(v, str) and v != ""

   186	def extract_assistant_text(message: Any) -> str:
   187	    """Pull human-readable text out of a message object.
   188	
   189	    Handles both shapes seen in this project's real data:
   190	      - message.content is a plain string
   191	      - message.content is a list of blocks like {"type": "text", "text": "..."}
   192	        (thinking blocks and anything without type == "text" are ignored)
   193	    Never raises; returns "" if nothing recognizable is found.
   194	    """
   195	    if not isinstance(message, dict):
   196	        return ""
   197	    content = message.get("content")
   198	    if isinstance(content, str):
   199	        return content
   200	    if isinstance(content, list):
   201	        parts = []
   202	        for block in content:
   203	            if isinstance(block, dict) and block.get("type") == "text":
   204	                text = block.get("text")
   205	                if isinstance(text, str):
   206	                    parts.append(text)
   207	        return "".join(parts)
   208	    return ""

   240	def parse_wire_trace(path: str) -> Tuple[Dict[Key, str], Dict[str, Any]]:
   241	    stats: Dict[str, Any] = {
   242	        "total_lines": 0,
   243	        "blank_lines": 0,
   244	        "parse_error_lines": 0,
   245	        "parse_error_line_numbers": [],
   246	        "unexpected_shape_lines": 0,
   247	        "unexpected_shape_line_numbers": [],
   248	        "no_delta_lines": 0,
   249	        "bad_role_delta_lines": 0,
   250	        "bad_role_delta_line_numbers": [],
   251	        "role_distribution": Counter(),
   252	        "malformed_candidate_lines": 0,
   253	        "malformed_line_numbers": [],
   254	        "matched_frames": 0,
   255	        "duplicate_keys": 0,
   256	    }
   257	    recs: Dict[Key, str] = {}
   258	
   259	    with open(path, "r", encoding="utf-8") as fh:
   260	        for lineno, raw_line in enumerate(fh, start=1):
   261	            stats["total_lines"] += 1
   262	            line = raw_line.strip()
   263	            if not line:
   264	                stats["blank_lines"] += 1
   265	                continue
   266	
   267	            try:
   268	                obj = json.loads(line)
   269	            except json.JSONDecodeError:
   270	                # Anomaly, not benign noise: a line that isn't valid JSON at all
   271	                # means the trace itself is suspect. Counted here and gated on
   272	                # in main() -- see anomaly_count below.
   273	                stats["parse_error_lines"] += 1
   274	                stats["parse_error_line_numbers"].append(lineno)
   275	                continue
   276	
   277	            if not isinstance(obj, dict):
   278	                stats["unexpected_shape_lines"] += 1
   279	                stats["unexpected_shape_line_numbers"].append(lineno)
   280	                continue
   281	
   282	            produced = obj.get("producedEvents")
   283	            has_delta = isinstance(produced, list) and any(
   284	                isinstance(e, dict) and e.get("wireType") == "evt.message.delta"
   285	                for e in produced
   286	            )
   287	            if not has_delta:
   288	                stats["no_delta_lines"] += 1
   289	                continue
   290	
   291	            role = dig(obj, "wireFrame", "payload", "message", "role")
   292	            stats["role_distribution"][str(role)] += 1
   293	            if role != "assistant":
   294	                # Anomaly, NOT benign filtering (2026-08-11 second-round fix):
   295	                # EventMapping.swift:149-177 (mapOpenclawSessionMessageToKernelEvents)
   296	                # is the ONLY place evt.message.delta is ever constructed, and it
   297	                # short-circuits to "no events" unless message.role == "assistant"
   298	                # (exact string match) BEFORE producing anything. So a wire line
   299	                # whose producedEvents[] already contains evt.message.delta is only
   300	                # consistent with wireFrame.payload.message.role being exactly
   301	                # "assistant" -- missing/non-string/any-other-value role here
   302	                # contradicts that already-observed fact and must fail closed.
   303	                # This is NOT the same rule as the history side: history's
   304	                # messages[] legitimately interleaves role == "user" entries
   305	                # (see extract_history_records below, where role != assistant
   306	                # stays normal semantics) -- the two sides are asymmetric on
   307	                # purpose and must not share this check.
   308	                stats["bad_role_delta_lines"] += 1
   309	                stats["bad_role_delta_line_numbers"].append(lineno)
   310	                continue
   311	
   312	            message_id = dig(obj, "wireFrame", "payload", "messageId")
   313	            message_seq = dig(obj, "wireFrame", "payload", "messageSeq")
   314	            message_obj = dig(obj, "wireFrame", "payload", "message") or {}
   315	
   316	            if not is_valid_id(message_id) or not is_valid_seq(message_seq):
   317	                # Anomaly, fail-closed (2026-08-11 second-round fix): see
   318	                # is_valid_id()/is_valid_seq() near the top of this file for
   319	                # why "is not None" wasn't enough (bool/float/str type
   320	                # confusion, e.g. JSON `true` aliasing integer `1`).

   109	/// 把一个内存里的 `Any`（来自 JSONSerialization 解出的 JSON 值）包成 D2 codegen 的 `JSONAny`。
   110	/// `JSONAny` 只暴露 `init(from: Decoder)`（Decodable 要求），没有直接take `Any` 的构造器——这里借道
   111	/// `JSONSerialization` + `JSONDecoder` 往返一次：包一层 `{"v": value}` 再解出来，是因为
   112	/// `JSONSerialization.data(withJSONObject:)` 在顶层只接受 Array/Dictionary，标量值（纯字符串/数字/
   113	/// 布尔）必须先包一层才能序列化。`value` 传 `NSNull()` 时往返出的是 JSON `null`——用于诚实表达"这个
   114	/// 字段本轮确实取不到真实值"（如非 exec 工具的 output，见②），不是遗漏，是明确的空值。
   115	func makeJSONAny(_ value: Any) -> JSONAny {
   116	    let wrapper: JSONObject = ["v": value]
   117	    guard JSONSerialization.isValidJSONObject(wrapper),
   118	          let data = try? JSONSerialization.data(withJSONObject: wrapper, options: []),
   119	          let decoded = try? JSONDecoder().decode([String: JSONAny].self, from: data),
   120	          let result = decoded["v"] else {
   121	        // "null" 恒可解码（走 JSONAny.init(from:) 的 singleValueContainer + decodeNil 分支），
   122	        // 用作不可序列化输入（例如非 JSON 兼容的 Swift 值）时的诚实兜底，不是伪造业务数据。
   123	        return try! JSONDecoder().decode(JSONAny.self, from: Data("null".utf8))
   124	    }
   125	    return result
   126	}

   166	func mapOpenclawSessionMessageToKernelEvents(
   167	    _ payload: JSONObject,
   168	    ourSessionID: String,
   169	    runIDHint: String?,
   170	    nextSeq: () -> Int
   171	) -> [EventMessageUnion] {
   172	    guard let message = jsonObject(payload["message"]) else { return [] }
   173	    guard jsonString(message["role"]) == "assistant" else {
   174	        // role=="user"：回显，无 D2 对应事件。role=="toolResult" 从未在 session.message 里现场
   175	        // 观察到（见头注释①），如果未来 openclaw 版本改变了这一点、这里会静默丢弃——保留在这里
   176	        // 一并说明，而不是假装处理了一个从未见过的分支。

   178	    }
   179	
   180	    let sentAt = Date()
   181	    let originTS = msEpochToDate(jsonInt(message["timestamp"]))
   182	    var events: [EventMessageUnion] = []
   183	
   184	    // 顶层字段（usage/stopReason/timestamp）是 assistant 消息自己的，不是某个 content block 的——
   185	    // 供 turnComplete 之类的调用方缓存 usage 用，这里只按 content block 逐个产出事件。

   187	    let blocks: [JSONObject]
   188	    if let arr = jsonArray(content) {
   189	        blocks = arr.compactMap { jsonObject($0) }
   190	    } else if let text = jsonString(content) {
   191	        // 纯字符串 content——SG-4 唯一覆盖过的形状，保留兼容。

   193	    } else {
   194	        blocks = []
   195	    }
   196	
   197	    for (index, block) in blocks.enumerated() {
   198	        guard let type = jsonString(block["type"]) else { continue }
   199	        switch type {

   209	            let deltaPayload = MessageDeltaEventMessagePayload(delta: text, index: index, messageID: messageID, role: .assistant)
   210	            events.append(.messageDelta(MessageDeltaEventMessage(
   211	                direction: .event, payload: deltaPayload, runID: runIDHint,
   212	                sentAt: sentAt, seq: nextSeq(), sessionID: ourSessionID, ts: originTS, type: .evtMessageDelta
   213	            )))
   214	
   215	        case "thinking", "reasoning", "redacted_thinking":
   216	            // grounding：源码——kernels/openclaw/src/gateway/chat-display-projection.ts
   217	            // `isAssistantInternalReasoningContentType` 把 "thinking"/"reasoning"/

   219	            // （第一个来源是下面 handleAgentEvent 的 `agent(stream:"thinking")`，F5 新增，
   220	            // 源码级 grounding：`embedded-agent-subscribe.ts:1226-1254` `emitReasoningStream`）——
   221	            // 两者不冲突：session.message 里的 thinking block 是"这条已落地的 assistant 消息带的

   223	            // 可能两条路径都会各自产出一次 D2 thinking 事件，这是 openclaw 自身双通道广播的忠实反映，
   224	            // 不是本文件重复映射的 bug。
   225	            let delta = jsonString(block["text"]) ?? jsonString(block["thinking"]) ?? ""
   226	            let visibility: Visibility = type == "redacted_thinking" ? .summary : .raw
   227	            let thinkingPayload = ThinkingEventMessagePayload(delta: delta, visibility: visibility)
   228	            events.append(.thinking(ThinkingEventMessage(

   240	            )))
   241	
   242	        default:
   243	            // 未识别的 content block type——如实跳过，不臆造映射。
   244	            continue
   245	        }
   246	    }
   247	
   248	    return events

   240	def parse_wire_trace(path: str) -> Tuple[Dict[Key, str], Dict[str, Any]]:
   241	    stats: Dict[str, Any] = {
   242	        "total_lines": 0,
   243	        "blank_lines": 0,
   244	        "parse_error_lines": 0,
   245	        "parse_error_line_numbers": [],
   246	        "unexpected_shape_lines": 0,
   247	        "unexpected_shape_line_numbers": [],
   248	        "no_delta_lines": 0,
   249	        "bad_role_delta_lines": 0,
   250	        "bad_role_delta_line_numbers": [],
   251	        "role_distribution": Counter(),
   252	        "malformed_candidate_lines": 0,
   253	        "malformed_line_numbers": [],
   254	        "matched_frames": 0,
   255	        "duplicate_keys": 0,
   256	    }
   257	    recs: Dict[Key, str] = {}
   258	
   259	    with open(path, "r", encoding="utf-8") as fh:
   260	        for lineno, raw_line in enumerate(fh, start=1):
   261	            stats["total_lines"] += 1
   262	            line = raw_line.strip()
   263	            if not line:
   264	                stats["blank_lines"] += 1
   265	                continue
   266	
   267	            try:
   268	                obj = json.loads(line)
   269	            except json.JSONDecodeError:
   270	                # Anomaly, not benign noise: a line that isn't valid JSON at all
   271	                # means the trace itself is suspect. Counted here and gated on
   272	                # in main() -- see anomaly_count below.
   273	                stats["parse_error_lines"] += 1
   274	                stats["parse_error_line_numbers"].append(lineno)
   275	                continue
   276	
   277	            if not isinstance(obj, dict):
   278	                stats["unexpected_shape_lines"] += 1
   279	                stats["unexpected_shape_line_numbers"].append(lineno)
   280	                continue
   281	
   282	            produced = obj.get("producedEvents")
   283	            has_delta = isinstance(produced, list) and any(
   284	                isinstance(e, dict) and e.get("wireType") == "evt.message.delta"
   285	                for e in produced
   286	            )
   287	            if not has_delta:
   288	                stats["no_delta_lines"] += 1
   289	                continue
   290	
   291	            role = dig(obj, "wireFrame", "payload", "message", "role")
   292	            stats["role_distribution"][str(role)] += 1
   293	            if role != "assistant":
   294	                # Anomaly, NOT benign filtering (2026-08-11 second-round fix):
   295	                # EventMapping.swift:149-177 (mapOpenclawSessionMessageToKernelEvents)
   296	                # is the ONLY place evt.message.delta is ever constructed, and it
   297	                # short-circuits to "no events" unless message.role == "assistant"
   298	                # (exact string match) BEFORE producing anything. So a wire line
   299	                # whose producedEvents[] already contains evt.message.delta is only
   300	                # consistent with wireFrame.payload.message.role being exactly
   301	                # "assistant" -- missing/non-string/any-other-value role here
   302	                # contradicts that already-observed fact and must fail closed.
   303	                # This is NOT the same rule as the history side: history's
   304	                # messages[] legitimately interleaves role == "user" entries
   305	                # (see extract_history_records below, where role != assistant
   306	                # stays normal semantics) -- the two sides are asymmetric on
   307	                # purpose and must not share this check.
   308	                stats["bad_role_delta_lines"] += 1
   309	                stats["bad_role_delta_line_numbers"].append(lineno)
   310	                continue
   311	
   312	            message_id = dig(obj, "wireFrame", "payload", "messageId")
   313	            message_seq = dig(obj, "wireFrame", "payload", "messageSeq")
   314	            message_obj = dig(obj, "wireFrame", "payload", "message") or {}
   315	
   316	            if not is_valid_id(message_id) or not is_valid_seq(message_seq):
   317	                # Anomaly, fail-closed (2026-08-11 second-round fix): see
   318	                # is_valid_id()/is_valid_seq() near the top of this file for
   319	                # why "is not None" wasn't enough (bool/float/str type
   320	                # confusion, e.g. JSON `true` aliasing integer `1`).
   321	                stats["malformed_candidate_lines"] += 1
   322	                stats["malformed_line_numbers"].append(lineno)
   323	                continue
   324	
   325	            key = (message_id, message_seq)
   326	            if key in recs:
   327	                stats["duplicate_keys"] += 1
   328	            stats["matched_frames"] += 1
   329	            # Later frames for the same key (streaming growth) presumably carry the
   330	            # fuller content, so last-writer-wins. This collapse is intentional and
   331	            # NOT an anomaly: one logical assistant message legitimately spans many
   332	            # delta frames sharing the same (messageId, messageSeq) key on the wire.
   333	            # Contrast with the history side (extract_history_records below), where
   334	            # each entry is already a materialized, distinct message and a repeated
   335	            # key has no such benign explanation.
   336	            recs[key] = extract_assistant_text(message_obj)
   337	
   338	    stats["anomaly_count"] = (
   339	        stats["parse_error_lines"] + stats["unexpected_shape_lines"]
   340	        + stats["bad_role_delta_lines"] + stats["malformed_candidate_lines"]
   341	    )
   342	    return recs, stats

   349	def fetch_history_http(
   350	    base: str, session_key: str, limit: int, cursor: Optional[str], token: str, timeout: float
   351	) -> Tuple[Any, str]:
   352	    """Fetch exactly one page. Returns (parsed_json, url) -- the url is returned
   353	    (never the token, which only ever lives in the Authorization header) so
   354	    callers can cite it in pagination/diagnostic messages."""
   355	    base_clean = base.rstrip("/")
   356	    key_seg = urllib.parse.quote(session_key, safe=":")
   357	    params: Dict[str, Any] = {"limit": limit}
   358	    if cursor is not None:
   359	        params["cursor"] = cursor
   360	    query = urllib.parse.urlencode(params)
   361	    url = "%s/sessions/%s/history?%s" % (base_clean, key_seg, query)
   362	
   363	    req = urllib.request.Request(url, method="GET")
   364	    req.add_header("Authorization", "Bearer %s" % token)
   365	    req.add_header("Accept", "application/json")
   366	
   367	    try:
   368	        with urllib.request.urlopen(req, timeout=timeout) as resp:
   369	            raw = resp.read()
   370	            status = resp.getcode()
   371	    except urllib.error.HTTPError as e:
   372	        body = ""
   373	        try:
   374	            body = e.read().decode("utf-8", errors="replace")[:500]
   375	        except Exception:
   376	            pass
   377	        # NOTE: url intentionally excludes the token (it only ever lives in the
   378	        # Authorization header, never in the URL), so it's safe to print.
   379	        raise ReconcileError(
   380	            "history HTTP 请求失败：HTTP %s %s | url=%s | body(截断500字符)=%r"
   381	            % (e.code, e.reason, url, body)
   382	        )
   383	    except urllib.error.URLError as e:
   384	        raise ReconcileError("history HTTP 请求失败：无法连接 %s（%s）" % (url, e.reason))
   385	    except OSError as e:
   386	        raise ReconcileError("history HTTP 请求失败：%s（url=%s）" % (e, url))
   387	
   388	    try:
   389	        return json.loads(raw.decode("utf-8")), url
   390	    except (json.JSONDecodeError, UnicodeDecodeError) as e:
   391	        raise ReconcileError("history 响应（HTTP %s）不是合法 JSON：%s" % (status, e))
   392

   394	def load_history_file(path: str) -> Any:
   395	    try:
   396	        with open(path, "r", encoding="utf-8") as fh:
   397	            return json.load(fh)
   398	    except FileNotFoundError:
   399	        raise ReconcileError("--history-file 不存在：%s" % path)
   400	    except json.JSONDecodeError as e:
   401	        raise ReconcileError("--history-file 不是合法 JSON（%s）：%s" % (path, e))

   404	def _validate_history_page(page_obj: Any, page_no: int, source_desc: str) -> Tuple[List[Any], bool, Optional[str]]:
   405	    """Validate one page's shape and completeness metadata.
   406	
   407	    Returns (messages, has_more, next_cursor). Used for both the online (HTTP,
   408	    one call per page) and offline (--history-file, always exactly one "page")
   409	    sources so both go through the identical completeness gate: a page whose
   410	    hasMore/nextCursor can't be trusted must never be silently treated as "the
   411	    whole history" -- if wire happens to be missing exactly the messages that
   412	    got left off this page, both assertions would otherwise pass by accident.
   413	    """
   414	    if not isinstance(page_obj, dict):
   415	        raise ReconcileError(
   416	            "history 响应顶层不是对象（%s，第 %d 页）：%s" % (source_desc, page_no, type(page_obj).__name__)
   417	        )
   418	    messages = page_obj.get("messages")
   419	    if not isinstance(messages, list):
   420	        raise ReconcileError(
   421	            "history 响应缺少合法的 'messages' 数组（%s，第 %d 页），无法对账" % (source_desc, page_no)
   422	        )
   423	    has_more = page_obj.get("hasMore")
   424	    if has_more is not True and has_more is not False:
   425	        raise ReconcileError(
   426	            "history 响应缺少合法的布尔 'hasMore' 字段（%s，第 %d 页，实际值=%r），"
   427	            "无法判断是否已经读全整个 history，拒绝在无法确认完整性的数据上对账（硬失败，不只是警告）"
   428	            % (source_desc, page_no, has_more)
   429	        )
   430	    next_cursor = page_obj.get("nextCursor")
   431	    if has_more and (not isinstance(next_cursor, str) or not next_cursor):
   432	        raise ReconcileError(
   433	            "history 响应 hasMore=true 但缺少合法的 'nextCursor'（%s，第 %d 页），无法继续翻页——"
   434	            "拒绝在不完整的数据上对账" % (source_desc, page_no)
   435	        )
   436	    return messages, bool(has_more), (next_cursor if has_more else None)
   437

   439	def resolve_offline_history(path: str) -> Tuple[List[Any], str]:
   440	    """Offline snapshots are necessarily a single page. Apply the same
   441	    completeness gate as the online path: hasMore must be explicitly false,
   442	    or we refuse to reconcile against a knowingly-incomplete snapshot (there's
   443	    no way to page further with just a static file)."""
   444	    history_obj = load_history_file(path)
   445	    messages, has_more, _next_cursor = _validate_history_page(history_obj, 1, "file:%s" % path)
   446	    if has_more:
   447	        raise ReconcileError(
   448	            "--history-file 离线快照的 hasMore=true——快照本身不完整（还有更早的消息没包含在这份快照"
   449	            "里），离线模式没法继续翻页，拒绝在残缺数据上对账（硬失败，不是警告）：%s" % path
   450	        )
   451	    return messages, "file:%s" % path

   454	def resolve_online_history(
   455	    base: str, session_key: str, limit: int, token: str, timeout: float
   456	) -> Tuple[List[Any], str]:
   457	    """Follow nextCursor and keep requesting pages until hasMore is false.
   458	    A truncated first page must never be silently treated as the whole
   459	    history: if the messages wire lost are exactly the ones a truncated page
   460	    would have left off, both assertions would otherwise pass by accident."""
   461	    all_messages: List[Any] = []
   462	    seen_cursors = set()
   463	    cursor: Optional[str] = None
   464	    page_no = 0
   465	    last_url = ""
   466	    source_desc = "%s/sessions/%s/history" % (base.rstrip("/"), session_key)
   467	    while True:
   468	        page_no += 1
   469	        if page_no > MAX_HISTORY_PAGES:
   470	            raise ReconcileError(
   471	                "history 翻页超过 %d 页仍未见 hasMore=false，判定分页失控（服务端缺陷或游标未推进），中止"
   472	                % MAX_HISTORY_PAGES
   473	            )
   474	        page_obj, last_url = fetch_history_http(base, session_key, limit, cursor, token, timeout)
   475	        messages, has_more, next_cursor = _validate_history_page(page_obj, page_no, source_desc)
   476	        all_messages.extend(messages)
   477	        if not has_more:
   478	            break
   479	        if next_cursor in seen_cursors:
   480	            raise ReconcileError(
   481	                "history 翻页出现重复的 nextCursor=%r（第 %d 页），游标没有推进，可能是服务端分页缺陷，"
   482	                "拒绝继续翻页" % (next_cursor, page_no)
   483	            )
   484	        seen_cursors.add(next_cursor)
   485	        cursor = next_cursor
   486	
   487	    label = "%s?limit=%s（在线翻页 %d 页直到 hasMore=false；最后一页 url=%s）" % (
   488	        source_desc, limit, page_no, last_url
   489	    )
   490	    return all_messages, label

   493	def extract_history_records(messages: List[Any]) -> Tuple[Dict[Key, str], Dict[str, Any]]:
   494	    """messages is already a validated, fully-paginated list (see resolve_*_history
   495	    above) -- this function no longer needs to guard against "not a list"."""
   496	    stats: Dict[str, Any] = {
   497	        "total_messages": 0,
   498	        "non_assistant_messages": 0,
   499	        "role_distribution": Counter(),
   500	        "non_dict_messages": 0,
   501	        "non_dict_indices": [],
   502	        "malformed_metadata_messages": 0,
   503	        "malformed_metadata_indices": [],
   504	        "duplicate_keys": 0,
   505	    }
   506	    recs: Dict[Key, str] = {}
   507	    key_texts: Dict[Key, List[str]] = {}
   508	
   509	    for idx, msg in enumerate(messages):
   510	        stats["total_messages"] += 1
   511	        if not isinstance(msg, dict):
   512	            # Anomaly: history entries are supposed to already be materialized
   513	            # message objects, not raw scalars/lists. Gated on in main().
   514	            stats["non_dict_messages"] += 1
   515	            stats["non_dict_indices"].append(idx)
   516	            continue
   517	
   518	        role = msg.get("role")
   519	        stats["role_distribution"][str(role)] += 1
   520	        if role != "assistant":
   521	            # Normal semantics, not an anomaly (e.g. user turns interleaved in
   522	            # the same history). Kept out of anomaly_count, but still visible
   523	            # via role_distribution above.
   524	            stats["non_assistant_messages"] += 1
   525	            continue
   526	
   527	        mid = dig(msg, "__openclaw", "id")
   528	        mseq = dig(msg, "__openclaw", "seq")
   529	        if not is_valid_id(mid) or not is_valid_seq(mseq):
   530	            # Anomaly, fail-closed (2026-08-11 second-round fix): see
   531	            # is_valid_id()/is_valid_seq() near the top of this file for why
   532	            # "is not None" wasn't enough (bool/float/str type confusion --
   533	            # a JSON boolean seq could otherwise alias an integer seq via
   534	            # Python's True == 1).
   535	            stats["malformed_metadata_messages"] += 1
   536	            stats["malformed_metadata_indices"].append(idx)
   537	            continue
   538	
   539	        key = (mid, mseq)
   540	        text = extract_assistant_text(msg)
   541	        if key in recs:
   542	            stats["duplicate_keys"] += 1
   543	        # Unlike the wire side, a repeated (id, seq) here has no benign
   544	        # explanation -- each entry in history's messages[] is supposed to be
   545	        # one already-materialized, distinct message, not a streaming delta.
   546	        # Overwriting silently (last-writer-wins) would hide the fact that two
   547	        # DISTINCT history entries collapsed onto one key, which is exactly
   548	        # the "history has 2, wire has 1, dict-collapse makes it look like a
   549	        # match" false-green this script now refuses to allow -- see
   550	        # duplicate_key_texts below, which main() turns into a hard failure.
   551	        recs[key] = text
   552	        key_texts.setdefault(key, []).append(text)
   553	
   554	    stats["malformed_messages"] = stats["non_dict_messages"] + stats["malformed_metadata_messages"]
   555	    stats["anomaly_count"] = stats["malformed_messages"]
   556	    stats["duplicate_key_texts"] = {k: v for k, v in key_texts.items() if len(v) > 1}
   557	    return recs, stats
   558

   560	# ---------------------------------------------------------------------------
   561	# reconciliation verdict -- single source of truth for pass/fail, reused by
   562	# normal mode and (unmodified) by both the --drop-one baseline check and the
   563	# post-drop check, so "passed" means the exact same thing everywhere.
   564	# ---------------------------------------------------------------------------
   565	
   566	def evaluate_reconciliation(
   567	    wire_recs: Dict[Key, str],
   568	    history_recs: Dict[Key, str],
   569	    history_duplicate_keys: List[Key],
   570	    wire_anomaly_count: int,
   571	    history_anomaly_count: int,
   572	    expect_min_assistant: int,
   573	) -> Dict[str, Any]:
   574	    wire_only = sorted(set(wire_recs) - set(history_recs))
   575	    history_only = sorted(set(history_recs) - set(wire_recs))
   576	    wire_count = len(wire_recs)
   577	    history_count = len(history_recs)
   578	    passed = (
   579	        not wire_only
   580	        and not history_only
   581	        and not history_duplicate_keys
   582	        and wire_anomaly_count == 0
   583	        and history_anomaly_count == 0
   584	        and wire_count >= expect_min_assistant
   585	        and history_count >= expect_min_assistant
   586	    )
   587	    return {
   588	        "wire_only": wire_only,
   589	        "history_only": history_only,
   590	        "history_duplicate_keys": history_duplicate_keys,
   591	        "wire_anomaly_count": wire_anomaly_count,
   592	        "history_anomaly_count": history_anomaly_count,
   593	        "wire_assistant_count": wire_count,
   594	        "history_assistant_count": history_count,
   595	        "expect_min_assistant": expect_min_assistant,
   596	        "passed": passed,
   597	    }

   602	    history_recs: Dict[Key, str], history_stats: Dict[str, Any],
   603	) -> None:
   604	    print("--- wire 侧解析 ---")
   605	    kv("总行数", wire_stats["total_lines"])
   606	    kv("空行（跳过，非异常）", wire_stats["blank_lines"])
   607	    kv("JSON 解析失败（异常，计入判定）", wire_stats["parse_error_lines"])
   608	    sub("解析失败的行号", wire_stats["parse_error_line_numbers"])
   609	    kv("JSON 顶层非对象（异常，计入判定）", wire_stats["unexpected_shape_lines"])
   610	    sub("该类行号", wire_stats["unexpected_shape_line_numbers"])

   613	    sub("该类行号", wire_stats["bad_role_delta_line_numbers"])
   614	    kv("delta 帧 role 分布", dict(wire_stats["role_distribution"]))
   615	    kv("缺失或类型不合法的 messageId/messageSeq（异常，计入判定）", wire_stats["malformed_candidate_lines"])
   616	    sub("该类行号", wire_stats["malformed_line_numbers"])
   617	    kv("匹配的 assistant delta 帧（去重前）", wire_stats["matched_frames"])
   618	    kv("去重后 unique (id, seq)", len(wire_recs))

   621	    print()
   622	
   623	    print("--- history 侧解析 ---")
   624	    kv("messages 总条数", history_stats["total_messages"])
   625	    kv("role != assistant（忽略，正常语义）", history_stats["non_assistant_messages"])
   626	    kv("消息 role 分布", dict(history_stats["role_distribution"]))
   627	    kv("顶层非对象条目（异常，计入判定）", history_stats["non_dict_messages"])
   628	    sub("该类索引", history_stats["non_dict_indices"])
   629	    kv("缺失或类型不合法的 __openclaw.id/seq（异常，计入判定）", history_stats["malformed_metadata_messages"])
   630	    sub("该类索引", history_stats["malformed_metadata_indices"])
   631	    kv("history 侧异常合计（非对象+缺metadata）", history_stats["anomaly_count"])
   632	    kv("去重后 unique (id, seq)", len(history_recs))
   633	    kv(
   634	        "同一 (id,seq) 出现>=2次的key数（异常，计入判定）",
   635	        len(history_stats["duplicate_key_texts"]),
   636	    )
   637	    if history_stats["duplicate_key_texts"]:
   638	        print("      history 侧同键重复明细（每条都原样列出，不做去重覆盖）：")
   639	        for k in sorted(history_stats["duplicate_key_texts"]):
   640	            texts = history_stats["duplicate_key_texts"][k]
   641	            print("        %s  出现 %d 次" % (fmt_key(k), len(texts)))
   642	            for i, t in enumerate(texts):
   643	                print("          [%d] text[:60]=%r" % (i, preview(t)))
   644	    print()

    83	## 一个必须记下的纠正：`kernelSessionID` ≠ history 的 `key`
    84	
    85	对账差点跑错对象。openclaw `sessions.create` 同时返回两个**独立字段**：
    86	
    87	- `key` —— 形如 `agent:main:dashboard:<uuid>`，**`GET /sessions/<key>/history` 要的是这个**
    88	- `sessionId` —— 另一个独立生成的 id
    89	
    90	而 `SessionHandle.kernelSessionID` 存的是 `sessionId ?? key`，实际几乎总是 `sessionId`。
    91	CLI 原先只打印 `kernelSessionID`——**照它去查 history 会查到一个不存在的会话**。
    92	现已改为打印 `[C] SESSION_KEY=` 并附一行说明二者区别。
    93	
    94	本轮实测两值：`key=agent:main:dashboard:1be7cfd9-...`，`sessionId=8f0a808f-...`，**确实不同**。

    96	## 主会话自查：一条被堵住的假绿路径，与一条**残余风险**
    97	
    98	对账最危险的失败模式不是「报红」，是「本该红却绿」。自查了 history 分页截断这条路径：
    99	
   100	**实测**：把 `--limit` 从 200 压到 **2**（少于该会话的 6 条消息）→
   101	wire 侧 3 条、history 侧只剩 1 条 → **断言① 被违反 2 条 → FAIL（exit 1）**。
   102	
   103	**结论**：截断使 history 变小，虽然让断言②（history ⊆ wire）更容易通过，
   104	但必然从**断言①**（wire ⊆ history）那一侧暴露。**两向断言的设计堵住了这条假绿路径。**
   105	
   106	### 残余风险（未消除，如实登记）
   107	
   108	脚本**不会**在 `返回条数 == limit` 时发出「history 可能被截断」的警告。
   109	存在一个理论上的窄缝：**当 wire 真正丢失的消息，恰好就是被截断掉的那几条最旧消息，
   110	且没有其他消息落在截断区**——此时两个断言可同时通过。
   111	
   112	- 本轮不构成问题：`--limit 200` 对 6 条消息，无截断。
   113	- **未修的原因**：发现时 ★审查闸正在只读评审这个文件，中途改动会使评审失效。
   114	- **建议修法**：`len(messages) >= limit` 时打印显式警告并计入报告头。
